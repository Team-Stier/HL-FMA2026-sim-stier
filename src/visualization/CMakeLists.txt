cmake_minimum_required(VERSION 3.18)
project(visualization)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
find_package(rviz_common REQUIRED)
find_package(rviz_default_plugins REQUIRED)
find_package(pluginlib REQUIRED)
find_package(std_msgs REQUIRED)
add_library(idle_follow_view SHARED src/idle_follow_view.cpp src/hud_display.cpp src/aerial_display.cpp)
target_compile_features(idle_follow_view PRIVATE cxx_std_17)
ament_target_dependencies(idle_follow_view rviz_common rviz_default_plugins pluginlib std_msgs)
pluginlib_export_plugin_description_file(rviz_common rviz_plugins.xml)
install(TARGETS idle_follow_view LIBRARY DESTINATION lib)
ament_python_install_package(${PROJECT_NAME})
install(PROGRAMS scripts/visualizer_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch rviz DESTINATION share/${PROJECT_NAME})
install(FILES ../../config/vehicle.yaml ../../config/runtime.yaml DESTINATION share/${PROJECT_NAME}/config)
ament_package()
