cmake_minimum_required(VERSION 3.18)
project(visualization)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
find_package(rviz_common REQUIRED)
find_package(rviz_default_plugins REQUIRED)
find_package(pluginlib REQUIRED)
find_package(std_msgs REQUIRED)
find_package(visualization_msgs REQUIRED)
find_package(interfaces REQUIRED)
add_library(idle_follow_view SHARED src/idle_follow_view.cpp src/hud_display.cpp src/aerial_display.cpp src/cell_display.cpp)
target_compile_features(idle_follow_view PRIVATE cxx_std_17)
ament_target_dependencies(idle_follow_view rviz_common rviz_default_plugins pluginlib std_msgs visualization_msgs interfaces)
pluginlib_export_plugin_description_file(rviz_common rviz_plugins.xml)
install(TARGETS idle_follow_view LIBRARY DESTINATION lib)
ament_python_install_package(${PROJECT_NAME})
install(PROGRAMS scripts/visualizer_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch rviz DESTINATION share/${PROJECT_NAME})
install(FILES ../../config/vehicle.yaml ../../config/runtime.yaml DESTINATION share/${PROJECT_NAME}/config)
include(CTest)
if(BUILD_TESTING)
    add_executable(test_marker_snapshot src/test_snapshot.cpp)
    target_compile_features(test_marker_snapshot PRIVATE cxx_std_17)
    target_compile_options(test_marker_snapshot PRIVATE -UNDEBUG)
    ament_target_dependencies(test_marker_snapshot rviz_common visualization_msgs)
    add_test(NAME marker_snapshot COMMAND test_marker_snapshot)
    find_package(tf2_msgs REQUIRED)
    add_executable(test_cell_display src/test_cell_display.cpp)
    target_compile_features(test_cell_display PRIVATE cxx_std_17)
    target_compile_options(test_cell_display PRIVATE -UNDEBUG)
    ament_target_dependencies(test_cell_display rviz_common rviz_default_plugins interfaces tf2_msgs)
    add_test(NAME cell_contract COMMAND test_cell_display --contract)
    add_test(NAME cell_display COMMAND test_cell_display)
    set_tests_properties(cell_display PROPERTIES SKIP_RETURN_CODE 77 TIMEOUT 60 RUN_SERIAL TRUE)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    foreach(check markers dynamic_markers signals_static)
        add_test(NAME ${check} COMMAND ${Python3_EXECUTABLE} -B -m visualization.test_${check})
        set_tests_properties(${check} PROPERTIES
            ENVIRONMENT "PYTHONPATH=${CMAKE_CURRENT_SOURCE_DIR}:$ENV{PYTHONPATH}")
    endforeach()
endif()
ament_package()
